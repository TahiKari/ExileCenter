cmake_minimum_required(VERSION 3.20)
project(ExileCenterCamera LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ExileCenterCamera SHARED Camera.cpp Forwarders.asm Camera.def)
target_compile_definitions(ExileCenterCamera PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ExileCenterCamera PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/WX>)
set_target_properties(ExileCenterCamera PROPERTIES OUTPUT_NAME "ExileCenter.Camera")
add_executable(CameraChecks CameraChecks.cpp Fixture.asm)
target_compile_definitions(CameraChecks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CameraChecks PRIVATE version)
enable_testing()
add_executable(CameraFixture CameraChecks.cpp Fixture.asm)
target_compile_definitions(CameraFixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CameraFixture PRIVATE ExileCenterCamera)
